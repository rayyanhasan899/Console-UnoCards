cmake_minimum_required(VERSION 3.16)
project(console_uno LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This project uses Windows console APIs; use a Windows C++ toolchain.")
endif()
add_executable(console-uno Console-UnoCards.cpp)
target_compile_features(console-uno PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(console-uno PRIVATE /W4 /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(structures_test tests/structures_test.cpp)
  target_compile_features(structures_test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(structures_test PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(structures_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME structures_test COMMAND structures_test)
endif()
